        ' an assignment
        order = 1
        Dim b = From x In xs
        REM a member call
        group.Add(b)
        Dim c = From x In xs
        ' a compound assignment
        take += 1
        Dim d = From x In xs
        ' a shift assignment
        skip <<= 1
        Dim e = From x In xs

        distinct()
        Dim f = From x In xs
        ' clauses that start like those statements
        Where (x > 0)
        Where -x < 0
        Select .Name
    End Sub
End Module
